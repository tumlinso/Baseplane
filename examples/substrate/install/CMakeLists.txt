cmake_minimum_required(VERSION 3.20)
project(baseplane_installed_consumer LANGUAGES CXX)
enable_testing()
option(BP_REQUIRE_INTEGRATED "Require all integrated installed host components" OFF)
option(BP_REQUIRE_CUDA "Require substrate CUDA execution (must reject host-only APIs)" OFF)
set(BP_CE_PACKAGE Cellerator CACHE STRING "Explicit installed CE package")
find_package(Baseplane CONFIG REQUIRED)
get_target_property(imported Baseplane::seq IMPORTED)
if(NOT imported)
    message(FATAL_ERROR "Consumer must use installed Baseplane::seq")
endif()
set(consumer_links Baseplane::seq)
if(BP_REQUIRE_INTEGRATED)
    if(NOT COMMAND baseplane_load_substrate)
        message(FATAL_ERROR "Installed Baseplane substrate module unavailable")
    endif()
    set(cuda_argument "")
    if(BP_REQUIRE_CUDA)
        set(cuda_argument REQUIRE_CUDA)
    endif()
    baseplane_load_substrate(COMPONENTS query hierarchy index reuse learning
        CE_PACKAGE "${BP_CE_PACKAGE}" ${cuda_argument})
    set(consumer_links Baseplane::query Baseplane::hierarchy Baseplane::index Baseplane::reuse Baseplane::learning)
elseif(BP_REQUIRE_CUDA)
    message(FATAL_ERROR "This qualification consumer is CPU-only")
endif()
add_executable(baseplaneInstalledConsumer consumer.cpp)
target_link_libraries(baseplaneInstalledConsumer PRIVATE ${consumer_links})
target_compile_definitions(baseplaneInstalledConsumer PRIVATE BP_REQUIRE_INTEGRATED=$<BOOL:${BP_REQUIRE_INTEGRATED}>)
target_compile_options(baseplaneInstalledConsumer PRIVATE -Wall -Wextra -Werror)
add_test(NAME baseplaneInstalledConsumer COMMAND baseplaneInstalledConsumer)
