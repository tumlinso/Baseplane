cmake_minimum_required(VERSION 3.20)
project(baseplane_hierarchy_consumer LANGUAGES CXX)
enable_testing()
get_filename_component(BP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CE_MOON_SOURCE_DIR "" CACHE PATH "Inspected Cellerator numerical provider source")
if(NOT CE_MOON_SOURCE_DIR)
    message(FATAL_ERROR "Specify CE_MOON_SOURCE_DIR for actual CE effects/lifting providers")
endif()
set(BASEPLANE_ENABLE_CUDA OFF CACHE BOOL "Focused host consumer" FORCE)
add_subdirectory("${BP_ROOT}" baseplane EXCLUDE_FROM_ALL)
add_subdirectory("${CE_MOON_SOURCE_DIR}" ce-provider EXCLUDE_FROM_ALL)
baseplane_add_representation_target(bp_hierarchy_provider COMPONENTS Cellerator::moonshot_effects)
add_executable(baseplaneHierarchyTest test_hierarchy.cpp)
target_link_libraries(baseplaneHierarchyTest PRIVATE bp_hierarchy_provider)
target_include_directories(baseplaneHierarchyTest SYSTEM PRIVATE "${CE_MOON_SOURCE_DIR}/include")
target_compile_options(baseplaneHierarchyTest PRIVATE -Wall -Wextra -Werror)
add_test(NAME baseplaneHierarchyTest COMMAND baseplaneHierarchyTest)
