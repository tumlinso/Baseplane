cmake_minimum_required(VERSION 3.20)
project(baseplane_sequence_composition LANGUAGES CXX)
enable_testing()
find_package(Baseplane CONFIG REQUIRED)
find_package(Cellerator CONFIG REQUIRED COMPONENTS effects moonshot_learning)
if(NOT Cellerator_BUILD_SOURCE_REVISION STREQUAL CE_CORE_COMMIT)
    message(FATAL_ERROR "Installed Cellerator core source revision mismatch: ${Cellerator_BUILD_SOURCE_REVISION}")
endif()
get_target_property(core_imported Cellerator::effects IMPORTED)
get_target_property(bp_imported Baseplane::seq IMPORTED)
if(NOT core_imported OR NOT bp_imported)
    message(FATAL_ERROR "Composition requires installed core targets")
endif()
baseplane_load_substrate(COMPONENTS query hierarchy index reuse learning)
get_filename_component(BP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(sequenceComposition consumer.cpp)
target_include_directories(sequenceComposition PRIVATE "${BP_ROOT}/examples/substrate/sequence_tool")
target_link_libraries(sequenceComposition PRIVATE Baseplane::query Baseplane::hierarchy
    Baseplane::index Baseplane::reuse Baseplane::learning Cellerator::effects)
target_compile_options(sequenceComposition PRIVATE -Wall -Wextra -Werror)
add_test(NAME sequenceComposition COMMAND sequenceComposition)
file(WRITE "${CMAKE_BINARY_DIR}/core-source.txt" "${Cellerator_BUILD_SOURCE_REVISION}\n")
