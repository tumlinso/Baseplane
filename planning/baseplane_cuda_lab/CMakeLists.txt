cmake_minimum_required(VERSION 3.20)
project(BaseplaneCudaLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LAB_CUDA "Compile the unpromoted CUDA prototype drivers" OFF)
option(LAB_SANITIZE_HOST "Address/undefined sanitizers for scalar references" OFF)
add_library(lab_reference host/reference.cc)
target_include_directories(lab_reference PUBLIC include)
target_compile_options(lab_reference PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lab_host tests/host_tests.cc)
target_link_libraries(lab_host PRIVATE lab_reference)
if(LAB_SANITIZE_HOST)
  target_compile_options(lab_reference PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(lab_host PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(lab_host PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME scalar_contracts COMMAND lab_host all)
if(LAB_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 70 CACHE STRING "V100 baseline")
  endif()
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 13)
    message(FATAL_ERROR "Use CUDA 12.9 for this sm_70/V100 lab; CUDA 13 dropped Volta offline compilation")
  endif()
  find_package(CUDAToolkit REQUIRED)
  add_executable(lab_cuda cuda/driver.cu)
  target_include_directories(lab_cuda PRIVATE cuda)
  target_link_libraries(lab_cuda PRIVATE lab_reference CUDA::cudart)
  set_target_properties(lab_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(lab_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo;-Xptxas=-v>)
  # No --use_fast_math: numerical differences must be measured before relaxing semantics.
endif()
