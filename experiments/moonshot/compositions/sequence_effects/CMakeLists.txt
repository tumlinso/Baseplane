cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(bp_moon_c01 LANGUAGES CXX)
  include(CTest)
  set(CE_MOON_SOURCE_DIR "" CACHE PATH "Explicit Cellerator moonshot provider source")
  if(NOT CE_MOON_SOURCE_DIR)
    message(FATAL_ERROR "Set CE_MOON_SOURCE_DIR to the inspected Cellerator provider source")
  endif()
  add_subdirectory("${CE_MOON_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cellerator-moonshot")
  get_filename_component(bp_moon_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
  add_library(bp_moon_c01_substrate INTERFACE)
  add_library(Baseplane::moonshot ALIAS bp_moon_c01_substrate)
  target_include_directories(bp_moon_c01_substrate INTERFACE "${bp_moon_root}/include")
  target_compile_features(bp_moon_c01_substrate INTERFACE cxx_std_17)
  target_link_libraries(bp_moon_c01_substrate INTERFACE Cellerator::moonshot)
endif()
if(NOT TARGET Cellerator::moonshot_effects)
  message(FATAL_ERROR "C01 requires Cellerator::moonshot_effects")
endif()
get_filename_component(bp_moon_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(bp_moon_c01_smoke tests/smoke.cpp)
target_include_directories(bp_moon_c01_smoke PRIVATE "${bp_moon_root}/families")
target_link_libraries(bp_moon_c01_smoke PRIVATE Baseplane::moonshot Cellerator::moonshot_effects)
if(BUILD_TESTING)
  add_test(NAME bp_moon_c01_smoke COMMAND bp_moon_c01_smoke)
endif()
