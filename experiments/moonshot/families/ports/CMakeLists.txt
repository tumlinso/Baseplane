cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(bp_moon_ports LANGUAGES CXX)
  enable_testing()
  set(CE_MOON_SOURCE_DIR "" CACHE PATH "Explicit Cellerator moonshot provider root")
  if(NOT CE_MOON_SOURCE_DIR)
    message(FATAL_ERROR "CE_MOON_SOURCE_DIR must name the committed Cellerator experimental provider root")
  endif()
  add_subdirectory("${CE_MOON_SOURCE_DIR}" "cellerator-provider")
endif()
if(NOT TARGET Cellerator::moonshot_mechanisms)
  message(FATAL_ERROR "CE-MOON-050 mechanisms target is required")
endif()
add_library(bp_moon_ports INTERFACE)
add_library(Baseplane::moonshot_ports ALIAS bp_moon_ports)
target_include_directories(bp_moon_ports INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
target_link_libraries(bp_moon_ports INTERFACE Cellerator::moonshot Cellerator::moonshot_mechanisms)
target_compile_features(bp_moon_ports INTERFACE cxx_std_17)
add_executable(bp_moon_ports_probe probe.cpp)
target_link_libraries(bp_moon_ports_probe PRIVATE Baseplane::moonshot_ports)
add_test(NAME bp_moon_ports_probe COMMAND bp_moon_ports_probe)
