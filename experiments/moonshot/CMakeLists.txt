cmake_minimum_required(VERSION 3.18)
project(baseplane_moonshot LANGUAGES CXX)
include(CTest)
set(CE_MOON_SOURCE_DIR "" CACHE PATH "Explicit Cellerator moonshot provider source directory")
if(NOT TARGET Cellerator::moonshot)
  if(CE_MOON_SOURCE_DIR)
    add_subdirectory("${CE_MOON_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cellerator-moonshot")
  else()
    find_package(CelleratorMoonshot CONFIG QUIET)
  endif()
endif()
if(NOT TARGET Cellerator::moonshot)
  message(FATAL_ERROR "Provide target Cellerator::moonshot or CE_MOON_SOURCE_DIR pointing to its isolated source directory")
endif()
add_library(bp_moonshot INTERFACE)
add_library(Baseplane::moonshot ALIAS bp_moonshot)
target_include_directories(bp_moonshot INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(bp_moonshot INTERFACE cxx_std_17)
target_link_libraries(bp_moonshot INTERFACE Cellerator::moonshot)
add_executable(bp_moon_host_smoke host/smoke.cpp)
target_link_libraries(bp_moon_host_smoke PRIVATE Baseplane::moonshot)
add_executable(bp_moon_demo host/demo.cpp)
target_link_libraries(bp_moon_demo PRIVATE Baseplane::moonshot)
if(BUILD_TESTING)
  add_test(NAME bp_moon_host_smoke COMMAND bp_moon_host_smoke)
endif()
option(BP_MOON_ENABLE_CUDA "Enable CUDA for experimental family targets" OFF)
if(BP_MOON_ENABLE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_ARCHITECTURES 70)
endif()
# Each family owns its local CMakeLists; adding one needs no shared build edit.
file(GLOB family_builds CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/families/*/CMakeLists.txt")
foreach(family_build IN LISTS family_builds)
  get_filename_component(family_dir "${family_build}" DIRECTORY)
  get_filename_component(family_name "${family_dir}" NAME)
  add_subdirectory("${family_dir}" "families/${family_name}")
endforeach()
# Compositions reuse the same substrate and providers; the child entry guards
# recursive inclusion when configured standalone through this parent project.
add_subdirectory(compositions)
